Python

from dataclasses import dataclass

@dataclass
class Post:
    title: str
    tags: list[str]
    draft: bool = False

    def __repr__(self) -> str:
        return f"Post({self.title!r})"

post = Post("Hello World", ["emacs", "nix"])
print(post)
